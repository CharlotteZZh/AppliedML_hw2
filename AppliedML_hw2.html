Step 2 Gradient Boosted Trees (GBDT)

Define the function

def run_gbdt_experiment(
    experiment_name,
    learning_rate=0.1,
    max_depth=3,
    subsample=0.8,
    reg_lambda=1.0
):
    model = xgb.XGBClassifier(
        n_estimators=3000,
        learning_rate=learning_rate,
        max_depth=max_depth,
        subsample=subsample,
        reg_lambda=reg_lambda,
        tree_method="hist",
        eval_metric="logloss",
        early_stopping_rounds=30,
        random_state=RANDOM_STATE,
        n_jobs=-1
    )
    start_time = time.perf_counter()

    model.fit(
        X_train_gbdt,
        y_train,
        eval_set=[
            (X_train_gbdt, y_train),
            (X_val_gbdt, y_val)
        ],
        verbose=False
    )
    fit_time = time.perf_counter() - start_time

    # Validation predictions
    val_probabilities = model.predict_proba(X_val_gbdt)[:, 1]
    # Validation metrics
    val_ap = average_precision_score(
        y_val, val_probabilities
    )
    val_logloss = log_loss(
        y_val, val_probabilities
    )
    results = {
        "experiment": experiment_name,
        "learning_rate": learning_rate,
        "max_depth": max_depth,
        "subsample": subsample,
        "reg_lambda": reg_lambda,
        "best_iteration": model.best_iteration,
        "validation_AP": val_ap,
        "validation_logloss": val_logloss,
        "fit_time_seconds": fit_time
    }
    return model, results

Step 3 Multi-Layer Perceptron (MLP)

Define a function

def run_mlp_experiment(
    experiment_name,
    hidden_layer_sizes=(128, 64),
    learning_rate_init=0.001,
    alpha=0.0001
):

    model = MLPClassifier(
        hidden_layer_sizes=hidden_layer_sizes,
        activation="relu",
        solver="adam",
        learning_rate_init=learning_rate_init,
        alpha=alpha,
        max_iter=300,          
        batch_size=1024,
        early_stopping=False,
        random_state=RANDOM_STATE
    )

    # Capture convergence warnings
    with warnings.catch_warnings(record=True) as caught_warnings:
        warnings.simplefilter("always", ConvergenceWarning)
        start_time = time.perf_counter()
        model.fit(X_train_mlp, y_train)
        fit_time = time.perf_counter() - start_time

    convergence_warning = any(
        issubclass(w.category, ConvergenceWarning)
        for w in caught_warnings
    )

    # Validation predictions
    val_probabilities = model.predict_proba(X_val_mlp)[:, 1]
    # Validation metrics
    val_ap = average_precision_score(
        y_val, val_probabilities
    )

    val_logloss = log_loss(
        y_val, val_probabilities
    )

    results = {
        "experiment": experiment_name,
        "hidden_layer_sizes": hidden_layer_sizes,
        "learning_rate_init": learning_rate_init,
        "alpha": alpha,
        "n_iter": model.n_iter_,
        "training_loss": model.loss_,
        "validation_AP": val_ap,
        "validation_logloss": val_logloss,
        "fit_time_seconds": fit_time,
        "convergence_warning": convergence_warning
    }
    return model, results
